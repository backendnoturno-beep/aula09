//arquivo usersModel.js

//chamada do método 'uuid' para criação de 'ID' para o usuário
const uuid = require('uuid').v4

//chamada do método 'bcrypt' para criar chave(token) criptografado (autenticação do usuário)
const bcrypt = require('bcrypt')

//dados para serem carregados na inicialiação
const users = [
  { id: '1', name: 'Luiz', email: 'luiz@email.com', password: bcrypt.hashSync('123456', 10) },
  { id: '2', name: 'Rassi', email: 'rassi@email.com', password: bcrypt.hashSync('0000', 10) }
]

//funções do CRUD para serem exportadas

module.exports = {

  createUser: (name, email, password) => {
    const newUser = {
      id: uuid(),
      name,
      email,
      password: bcrypt.hashSync(password, 10)
    }
    users.push(newUser)
    return newUser
  },

  getAllUsers: () => users,

  getUserById: (id) => users.find(user => user.id === id),

  getUserByEmail: (email) => users.find(user => user.email === email),

  updateUser: (id, name, email, password) => {
    const user = users.find(u => u.id === id)
    if (!user) return null

    if (name) user.name = name
    if (email) user.email = email
    if (password) {
      user.password = bcrypt.hashSync(password, 10)
    }

    return user
  },

  deleteUser: (id) => {
    const index = users.findIndex(user => user.id === id)
    if (index === -1) return null

    // Remove o usuário do array e retorna o usuário excluído
    const deletedUser = users.splice(index, 1)
    return deletedUser[0]
  }

}

//arquivo authController.js

//chamada do método 'jwt' para uso do token de autenticação
const jwt = require('jsonwebtoken')

//chamada do método 'bcrypt' para verificação do token de autenticação
const bcrypt = require('bcrypt')

const usersModel = require("../models/usersModel")

module.exports = {
  // POST /login
  login: (req, res) => {
    const { email, password } = req.body

    if (typeof email !== 'string' || typeof password !== 'string') {
      return res.status(400).json({ message: 'Todos os campos são obrigatórios' })
    }

    const user = usersModel.getUserByEmail(email)
    if (!user) {
      return res.status(404).json({ message: 'Usuário não encontrado!' })
    }

    const isValidPassword = bcrypt.compareSync(password, user.password)
    if (!isValidPassword) {
      return res.status(401).json({ message: 'Credenciais incorretas!' })
    }

    const payload = { id: user.id, email: user.email }
    const token = jwt.sign(payload, process.env.JWT_KEY, { expiresIn: '1d' })
    res.json({ token })
  },

  // POST /cria
  create: (req, res) => {
    const { name, email, password } = req.body

    if (typeof name !== 'string' || typeof email !== 'string' || typeof password !== 'string') {
      return res.status(400).json({ message: 'Todos os campos são obrigatórios' })
    }

    const existingUser = usersModel.getUserByEmail(email)
    if (existingUser) {
      return res.status(400).json({ message: 'E-mail já cadastrado!' })
    }

    const newUser = usersModel.createUser(name, email, password)
    // res.status(201).json(newUser)
    res.status(201).json({ ...newUser, password: undefined })
  },

  //PUT /altera
  update: (req, res) => {
    const { id } = req.params
    const { name, email, password } = req.body
  
    const user = usersModel.getUserById(id)
    if (!user) {
      return res.status(404).json({ message: 'Usuário não encontrado!' })
    }
  
    // Verifica se o novo e-mail já pertence a outro usuário
    if (email && email !== user.email) {
      const existingUser = usersModel.getUserByEmail(email)
      if (existingUser) {
        return res.status(400).json({ message: 'E-mail já está em uso por outro usuário!' })
      }
    }
  
    const updatedUser = usersModel.updateUser(id, name, email, password)
  
    return res.json({
      message: 'Usuário atualizado com sucesso!',
      user: {
        id: updatedUser.id,
        name: updatedUser.name,
        email: updatedUser.email
      }
    })
  },

  // DELETE
  deleteUser: (req, res) => {
    const { id } = req.params
  
    const user = usersModel.getUserById(id)
    if (!user) {
      return res.status(404).json({ message: 'Usuário não encontrado!' })
    }
  
    usersModel.deleteUser(id)
    return res.json({ message: 'Usuário excluído com sucesso!', user: { id: user.id, name: user.name } })
  }

}

13 -  Crie o arquivo 'authMiddleware.js' dentro da pasta 'middlewares';

**************************************************************************************************************************************************************

//arquivo authMiddleware.js

//chamada do método 'jwt' para uso do token de autenticação
const jwt = require('jsonwebtoken')
const usersModel = require('../models/usersModel')

module.exports = {
  ensureAuth: (req, res, next) => {
    const authHeader = req.headers.authorization

    if (!authHeader) return res.status(401).json({ message: 'Não autorizado!' })

    const token = authHeader.split(' ')[1]

    try {
      const { id } = jwt.verify(token, process.env.JWT_KEY)
      const user = usersModel.getUserById(id)
      if (!user) return res.status(404).json({ message: 'Usuário não encontrado!' })
      req.user = user
      next()
    } catch (error) {
      return res.status(401).json({ message: 'Token inválido!' })
    }
  }
}

//arquivo auth.js"

const express = require('express')
const authController = require('../controllers/authController')
const { ensureAuth } = require('../middlewares/authMiddleware')
const usersModel = require('../models/usersModel')
const authRouter = express.Router()

authRouter.post('/login', authController.login)

authRouter.post('/user', authController.create)

authRouter.get('/user', ensureAuth, (req, res) => {
    const users = usersModel.getAllUsers()

    const safeUsers = users.map(user => ({
        id: user.id,
        name: user.name,
        email: user.email
    }))

    return res.json(safeUsers)
})

// Rota PUT para atualizar o usuário pelo ID
authRouter.put('/user/:id', ensureAuth, authController.update)

// Rota DELETE para excluir o usuário pelo ID
authRouter.delete('/user/:id', ensureAuth, authController.deleteUser)

module.exports = authRouter
